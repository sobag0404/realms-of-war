    10→// ─── Types ───────────────────────────────────────────────────────────────────
    11→
    12→/** Model definition using primitive shapes */
    13→export interface ModelDefinition {
    14→  id: string;
    15→  /** Shape type for primitive generation */
    16→  shape: 'box' | 'sphere' | 'cylinder' | 'cone' | 'compound';
    17→  /** Dimensions (width = X, height = Y, depth = Z) */
    18→  dimensions: { width: number; height: number; depth: number };
    19→  /** Base color as CSS string */
    20→  color: string;
    21→  /** Emissive color (subtle glow, optional) */
    22→  emissive?: string;
    23→  /** Additional shapes for compound models */
    24→  children?: Array<{
    25→    shape: 'box' | 'sphere' | 'cylinder' | 'cone';
    26→    position: [number, number, number];
    27→    dimensions: { width: number; height: number; depth: number };
    28→    color: string;
    29→    emissive?: string;
    30→  }>;
    31→}

   497→/** Create a BufferGeometry for a given primitive shape. */
   498→function createPrimitiveGeometry(
   499→  shape: 'box' | 'sphere' | 'cylinder' | 'cone',
   500→  dimensions: { width: number; height: number; depth: number },
   501→): THREE.BufferGeometry {
   502→  switch (shape) {
   503→    case 'box':
   504→      return new THREE.BoxGeometry(dimensions.width, dimensions.height, dimensions.depth);
   505→    case 'sphere':
   506→      return new THREE.SphereGeometry(Math.max(dimensions.width, dimensions.height) / 2, 12, 8);
   507→    case 'cylinder':
   508→      return new THREE.CylinderGeometry(
   509→        dimensions.width / 2,
   510→        dimensions.depth / 2,
   511→        dimensions.height,
   512→        12,
   513→      );
   514→    case 'cone':
   515→      return new THREE.ConeGeometry(
   516→        Math.max(dimensions.width, dimensions.depth) / 2,
   517→        dimensions.height,
   518→        12,
   519→      );
   520→    default: {
   521→      const _exhaustive: never = shape;
   522→      return new THREE.BoxGeometry(0.1, 0.1, 0.1);
   523→    }
   524→  }
   525→}
   526→
   527→/** Build a Three.js geometry from a model definition (top-level shape only). */
   528→export function buildGeometry(def: ModelDefinition): THREE.BufferGeometry {
   529→  return createPrimitiveGeometry(def.shape as 'box' | 'sphere' | 'cylinder' | 'cone', def.dimensions);
   530→}
   531→
   532→/** Build a Three.js material from a model definition. */
   533→export function buildMaterial(def: ModelDefinition): THREE.MeshStandardMaterial {
   534→  const params: THREE.MeshStandardMaterialParameters = {
   535→    color: new THREE.Color(def.color),
   536→    roughness: 0.7,
   537→    metalness: 0.1,
   538→    flatShading: true,
   539→  };
   540→  if (def.emissive) {
   541→    params.emissive = new THREE.Color(def.emissive);
   542→    params.emissiveIntensity = 0.3;
   543→  }
   544→  return new THREE.MeshStandardMaterial(params);
   545→}
   546→
   547→/** Build a complete Three.js mesh (or group) from a model definition. */
   548→export function buildMesh(def: ModelDefinition): THREE.Mesh | THREE.Group {
   549→  // Simple shape — single mesh
   550→  if (def.shape !== 'compound' || !def.children || def.children.length === 0) {
   551→    const geometry = createPrimitiveGeometry(
   552→      def.shape as 'box' | 'sphere' | 'cylinder' | 'cone',
   553→      def.dimensions,
   554→    );
   555→    const material = buildMaterial(def);
   556→    return new THREE.Mesh(geometry, material);
   557→  }
   558→
   559→  // Compound — group of child meshes
   560→  const group = new THREE.Group();
   561→
   562→  // Add children
   563→  for (const child of def.children) {
   564→    const childGeo = createPrimitiveGeometry(child.shape, child.dimensions);
   565→    const childMatParams: THREE.MeshStandardMaterialParameters = {
   566→      color: new THREE.Color(child.color),
   567→      roughness: 0.7,
   568→      metalness: 0.1,
   569→      flatShading: true,
   570→    };
   571→    if (child.emissive) {
   572→      childMatParams.emissive = new THREE.Color(child.emissive);
   573→      childMatParams.emissiveIntensity = 0.3;
   574→    }
   575→    const childMat = new THREE.MeshStandardMaterial(childMatParams);
   576→    const childMesh = new THREE.Mesh(childGeo, childMat);
   577→    childMesh.position.set(...child.position);
   578→    childMesh.castShadow = true;
   579→    childMesh.receiveShadow = true;
   580→    group.add(childMesh);
   581→  }
   582→
   583→  return group;
   584→}
