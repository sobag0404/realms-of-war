   497→   497→/** Create a BufferGeometry for a given primitive shape. */
   498→   498→function createPrimitiveGeometry(
   499→   499→  shape: 'box' | 'sphere' | 'cylinder' | 'cone',
   500→   500→  dimensions: { width: number; height: number; depth: number },
   501→   501→): THREE.BufferGeometry {
   502→   502→  switch (shape) {
   503→   503→    case 'box':
   504→   504→      return new THREE.BoxGeometry(dimensions.width, dimensions.height, dimensions.depth);
   505→   505→    case 'sphere':
   506→   506→      return new THREE.SphereGeometry(Math.max(dimensions.width, dimensions.height) / 2, 12, 8);
   507→   507→    case 'cylinder':
   508→   508→      return new THREE.CylinderGeometry(
   509→   509→        dimensions.width / 2,
   510→   510→        dimensions.depth / 2,
   511→   511→        dimensions.height,
   512→   512→        12,
   513→   513→      );
   514→   514→    case 'cone':
   515→   515→      return new THREE.ConeGeometry(
   516→   516→        Math.max(dimensions.width, dimensions.depth) / 2,
   517→   517→        dimensions.height,
   518→   518→        12,
   519→   519→      );
   520→   520→    default: {
   521→   521→      const _exhaustive: never = shape;
   522→   522→      return new THREE.BoxGeometry(0.1, 0.1, 0.1);
   523→   523→    }
   524→   524→  }
   525→   525→}
   526→   526→
   527→   527→/** Build a Three.js geometry from a model definition (top-level shape only). */
   528→   528→export function buildGeometry(def: ModelDefinition): THREE.BufferGeometry {
   529→   529→  return createPrimitiveGeometry(def.shape as 'box' | 'sphere' | 'cylinder' | 'cone', def.dimensions);
   530→   530→}
   531→   531→
   532→   532→/** Build a Three.js material from a model definition. */
   533→   533→export function buildMaterial(def: ModelDefinition): THREE.MeshStandardMaterial {
   534→   534→  const params: THREE.MeshStandardMaterialParameters = {
   535→   535→    color: new THREE.Color(def.color),
   536→   536→    roughness: 0.7,
   537→   537→    metalness: 0.1,
   538→   538→    flatShading: true,
   539→   539→  };
   540→   540→  if (def.emissive) {
   541→   541→    params.emissive = new THREE.Color(def.emissive);
   542→   542→    params.emissiveIntensity = 0.3;
   543→   543→  }
   544→   544→  return new THREE.MeshStandardMaterial(params);
   545→   545→}
   546→   546→
   547→   547→/** Build a complete Three.js mesh (or group) from a model definition. */
   548→   548→export function buildMesh(def: ModelDefinition): THREE.Mesh | THREE.Group {
   549→   549→  // Simple shape — single mesh
   550→   550→  if (def.shape !== 'compound' || !def.children || def.children.length === 0) {
   551→   551→    const geometry = createPrimitiveGeometry(
   552→   552→      def.shape as 'box' | 'sphere' | 'cylinder' | 'cone',
   553→   553→      def.dimensions,
   554→   554→    );
   555→   555→    const material = buildMaterial(def);
   556→   556→    return new THREE.Mesh(geometry, material);
   557→   557→  }
   558→   558→
   559→   559→  // Compound — group of child meshes
   560→   560→  const group = new THREE.Group();
   561→   561→
   562→   562→  // Add children
   563→   563→  for (const child of def.children) {
   564→   564→    const childGeo = createPrimitiveGeometry(child.shape, child.dimensions);
   565→   565→    const childMatParams: THREE.MeshStandardMaterialParameters = {
   566→   566→      color: new THREE.Color(child.color),
   567→   567→      roughness: 0.7,
   568→   568→      metalness: 0.1,
   569→   569→      flatShading: true,
   570→   570→    };
   571→   571→    if (child.emissive) {
   572→   572→      childMatParams.emissive = new THREE.Color(child.emissive);
   573→   573→      childMatParams.emissiveIntensity = 0.3;
   574→   574→    }
   575→   575→    const childMat = new THREE.MeshStandardMaterial(childMatParams);
   576→   576→    const childMesh = new THREE.Mesh(childGeo, childMat);
   577→   577→    childMesh.position.set(...child.position);
   578→   578→    childMesh.castShadow = true;
   579→   579→    childMesh.receiveShadow = true;
   580→   580→    group.add(childMesh);
   581→   581→  }
   582→   582→
   583→   583→  return group;
   584→   584→}
